    84	function pointDtoStyle(pointRole: 'free' | 'dependent', planeId: string, ctx: ReturnType<typeof stateFor>): { style: ResolvedStyle; layer: RenderLayer; radius: number } {
    85	  let fill: string; let stroke: string; let strokeWidth: number;
    86	  if (ctx.cascade) { fill = '#FF9800'; stroke = '#F57C00'; strokeWidth = 2.5; }
    87	  else if (ctx.snapTarget) { fill = '#FFA500'; stroke = '#FF8C00'; strokeWidth = 2; }
    88	  else if (ctx.selected) { fill = pointRole === 'free' ? '#4A90E2' : '#E94E77'; stroke = '#0066CC'; strokeWidth = 2; }
    89	  else if (ctx.hovered) { fill = pointRole === 'free' ? '#6BB6FF' : '#FF6B9D'; stroke = '#4A90E2'; strokeWidth = 1.5; }
    90	  else { fill = pointRole === 'free' ? '#4A90E2' : '#E94E77'; stroke = '#333333'; strokeWidth = 1.5; }
    91	  const zBoost = ctx.activePlaneId && planeId === ctx.activePlaneId ? 100 : 0;
    92	  const opacity = ctx.activePlaneId && planeId !== ctx.activePlaneId ? 0.5 : 1;
    93	  return { layer: ctx.snapTarget ? RenderLayer.Highlights : RenderLayer.Elements, radius: pointRole === 'free' ? 5 : 4, style: { stroke, strokeWidth, strokeStyle: 'solid', fill, opacity, zIndex: 10 + zBoost } };

    18	function getCanvasPlaneTransforms(world: World): CanvasPlaneTransformModel[] {
    19	  return world.getAllProjectionPlanes().flatMap((plane) => {
    20	    const realization = world.getProjectionPlane2DRealization(plane.id);
    21	    if (!realization) return [];
    22	    return [{ planeId: plane.id, planeToPaper: realization.planeToPaper2D, paperToPlane: realization.paperToPlane2D }];
    23	  });
    24	}

    26	function getPlaneCanvasItems(world: World, context: ProjectionElementRealizationContext): CanvasPlaneCanvasItems[] {
    27	  return world.getAllProjectionPlanes().flatMap((plane) => {
    28	    const transform = getCanvasPlaneTransforms(world).find(item => item.planeId === plane.id);
    29	    if (!transform) return [];
    30	    const items: CanvasPlaneItem[] = [];
    31	
    32	    for (const point of plane.getAllPoints()) {
    33	      if (!resolveVisibility(point, world)) continue;
    34	      const realization = plane.realizePoint2D(point, context);
    35	      if (!realization) continue;
    36	      items.push({
    37	        kind: 'point',
    38	        elementId: point.id,
    39	        planeId: point.planeId,
    40	        position: realization.position.clone(),
    41	        pointRole: point.constructor.name === 'FreePoint2D' ? 'free' : 'dependent',
    42	      });
    43	    }
    44	
    45	    for (const line of plane.getAllLines()) {
    46	      if (!resolveVisibility(line, world)) continue;
    47	      const realization = plane.realizeLine2D(line, context);
    48	      if (!realization) continue;
    49	      const kind: CanvasPlaneLineItem['kind'] = line instanceof HalfHingeLine2D
    50	        ? 'half-hinge-line'
    51	        : line instanceof Segment2DByPoints
    52	          ? 'finite-segment'
    53	          : line instanceof SiblingGuide2D
    54	            ? 'sibling-guide'
    55	            : line instanceof Guide2D
    56	              ? 'guide'
    57	              : 'infinite-line';
    58	      items.push({
    59	        kind,
    60	        elementId: line.id,
    61	        planeId: line.planeId,
    62	        point: realization.point1.clone(),
    63	        direction: realization.getDirection(),
    64	        point1: kind === 'finite-segment' ? realization.point1.clone() : undefined,
    65	        point2: kind === 'finite-segment' ? realization.point2.clone() : undefined,
    66	        arcGeometry: optionalArcGeometry(line),
    67	      });
    68	    }
    69	
    70	    for (const circle of plane.getAllCircles()) {
    71	      if (!resolveVisibility(circle, world)) continue;
    72	      const realization = plane.realizeCircle2D(circle, context);
    73	      if (!realization) continue;
    74	      items.push({
    75	        kind: 'circle',
    76	        elementId: circle.id,
    77	        planeId: circle.planeId,
    78	        center: realization.center.clone(),
    79	        radius: realization.getRadius(),
    80	        radiusVector: realization.radiusVector.clone(),
    81	      });
    82	    }
    83	
    84	    return [{ planeId: plane.id, transform, items }];
    85	  });
    86	}

     1	/**
     2	 * ProjectService - current-schema command-stream persistence orchestration.
     3	 */
     4	
     5	import { World } from '@/domain/world/World';
     6	import type { CommandStack } from '@/commands/CommandStack';
     7	import { Command, type CommandSnapshot } from '@/commands/Command';
     8	import type { OperationDescriptor } from '@/commands/operations/IOperation';
     9	import { deserializeOperation } from '@/commands/operations/OperationRegistry';
    10	
    11	export interface ProjectData {
    12	  id: string;
    13	  name: string;
    14	  schemaVersion: string;
    15	  created: string;
    16	  modified: string;
    17	  nextElementId: number;
    18	  commands: SerializedCommand[];
    19	}
    20	
    21	type SerializedCommand = CommandSnapshot;
    22	type SerializedOperation = OperationDescriptor;
    23	
    24	export class ProjectService {
    25	  private static readonly SCHEMA_VERSION = '6.0.0';
    26	
    27	  static serializeCommands(commandStack: CommandStack, nextElementId: number, projectName: string): ProjectData {
    28	    if (!Number.isInteger(nextElementId) || nextElementId < 1) {
    29	      throw new Error(`Invalid next element ID: ${nextElementId}`);
    30	    }
    31	    const now = new Date().toISOString();
    32	    const commands = commandStack.getCommittedCommands().map(command => command.toSnapshot());
    33	
    34	    return {
    35	      id: this.generateProjectId(),
    36	      name: projectName,
    37	      schemaVersion: this.SCHEMA_VERSION,
    38	      created: now,
    39	      modified: now,
    40	      nextElementId,
    41	      commands
    42	    };
    43	  }
    44	
    45	  static deserializeCommands(data: ProjectData, world: World, commandStack: CommandStack): void {
    46	    this.validateProjectData(data);
    47	    commandStack.clear();
    48	    world.restoreNextId(data.nextElementId);
    49	
    50	    const restoredCommands: Command[] = [];
    51	    for (const serializedCommand of data.commands) {
    52	      const operations = serializedCommand.operations.map((descriptor: SerializedOperation) => deserializeOperation(descriptor));
    53	      for (const operation of operations) {
    54	        const result = operation.apply(world);
    55	        if (!result.success) {
    56	          throw new Error(`Failed to replay operation ${operation.type}: ${result.error}`);
    57	        }
    58	      }
    59	      restoredCommands.push(Command.restore({
    60	        id: serializedCommand.id,
    61	        timestamp: serializedCommand.timestamp,
    62	        description: serializedCommand.description,
    63	        operations
    64	      }, world));
    65	    }
    66	
    67	    commandStack.restoreCommittedCommands(restoredCommands);
    68	
    69	    if (!world.getActivePlaneId()) {
    70	      const planes = world.getAllProjectionPlanes();
    71	      const ph = planes.find(p => p.id === 'projPlaneH' || p.name === 'P_H');
    72	      if (ph) {
    73	        world.setActivePlaneId(ph.id);
    74	      } else if (planes.length > 0) {
    75	        world.setActivePlaneId(planes[0].id);
    76	      }
    77	    }
    78	  }
    79	
    80	  static validateProjectData(data: unknown): asserts data is ProjectData {
    81	    if (!data || typeof data !== 'object' || Array.isArray(data)) {
    82	      throw new Error('Invalid project file - not a valid JSON object');
    83	    }
    84	    const record = data as Partial<ProjectData>;
    85	    if (record.schemaVersion !== this.SCHEMA_VERSION) {
    86	      throw new Error(`Unsupported project schema version: ${String(record.schemaVersion)}`);
    87	    }
    88	    for (const field of ['id', 'name', 'created', 'modified'] as const) {
    89	      if (typeof record[field] !== 'string' || record[field]!.length === 0) {
    90	        throw new Error(`Invalid project file - missing required field: ${field}`);
    91	      }
    92	    }
    93	    if (typeof record.nextElementId !== 'number' || !Number.isInteger(record.nextElementId) || record.nextElementId < 1) {
    94	      throw new Error('Invalid project file - nextElementId must be a positive integer');
    95	    }
    96	    if (!Array.isArray(record.commands)) {
    97	      throw new Error('Invalid project file - commands must be an array');
    98	    }
    99	    for (const command of record.commands) {
   100	      this.validateSerializedCommand(command);
   101	    }
   102	  }
   103	
   104	  private static validateSerializedCommand(command: unknown): asserts command is SerializedCommand {
   105	    if (!command || typeof command !== 'object' || Array.isArray(command)) {
   106	      throw new Error('Invalid project file - command must be an object');
   107	    }
   108	    const record = command as Partial<SerializedCommand>;
   109	    if (typeof record.id !== 'string' || record.id.length === 0) {
   110	      throw new Error('Invalid project file - command missing id');
   111	    }
   112	    if (typeof record.timestamp !== 'string' || Number.isNaN(Date.parse(record.timestamp))) {
   113	      throw new Error(`Invalid project file - command ${record.id} has invalid timestamp`);
   114	    }
   115	    if (typeof record.description !== 'string' || record.description.length === 0) {
   116	      throw new Error(`Invalid project file - command ${record.id} missing description`);
   117	    }
   118	    if (record.status !== 'committed') {
   119	      throw new Error(`Invalid project file - command ${record.id} must be committed`);
   120	    }
   121	    if (!Array.isArray(record.operations) || record.operations.length === 0) {
   122	      throw new Error(`Invalid project file - command ${record.id} must contain operations`);
   123	    }
   124	    for (const operation of record.operations) {
   125	      if (!operation || typeof operation !== 'object' || Array.isArray(operation) || typeof (operation as { opType?: unknown }).opType !== 'string') {
   126	        throw new Error(`Invalid project file - command ${record.id} has invalid operation descriptor`);
   127	      }
   128	    }
   129	  }
   130	
   131	  private static generateProjectId(): string {
   132	    return `proj_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`;
   133	  }
