    30	function directionFromPlane(direction2D: Vector3, matrix: number[]): Vector3 {
    31	  const x = matrix[0] * direction2D.x + matrix[1] * direction2D.y;
    32	  const y = matrix[3] * direction2D.x + matrix[4] * direction2D.y;
    33	  const z = matrix[6] * direction2D.x + matrix[7] * direction2D.y;

    37	function intersections(models: ThreeGuideIntersectionReadModel, planePoint: Vector4, planeNormal: Vector3, guideId: string): GuidePresentationDto[] {
    38	  const items: GuidePresentationDto[] = [];
    39	  const normal = { x: planeNormal.x, y: planeNormal.y, z: planeNormal.z };
    40	  for (const point of models.points) {
    41	    if (!isVisible3D(point) || !point.cartesian) continue;
    42	    const [x, y, z] = point.cartesian;
    43	    const position = new Vector4(x, y, z, 1);
    44	    if (isPointOnPlane3D(position, planePoint, normal)) items.push(marker(`${guideId}-point-${point.id}`, point.id, position));
    45	  }
    46	  for (const line of models.lines) {
    47	    if (!isVisible3D(line) || !line.point || !line.direction) continue;
    48	    const hit = intersectLinePlane3D(line.point, line.direction, planePoint, normal);
    49	    if (hit) items.push(marker(`${guideId}-line-${line.id}`, line.id, hit));
    50	  }
    51	  for (const segment of models.segments) {
    52	    if (!isVisible3D(segment) || !segment.startPoint || !segment.endPoint) continue;
    53	    const hit = intersectSegmentPlane3D(segment.startPoint, segment.endPoint, planePoint, normal);
    54	    if (hit) items.push(marker(`${guideId}-segment-${segment.id}`, segment.id, hit));
    55	  }
    56	  return items;
    57	}

    43	function fallbackGrid(): Segment2D[] {
    44	  const segments: Segment2D[] = [];
    45	  const halfSize = PLANE_RENDERING_CONFIG.FALLBACK_GRID.HALF_SIZE;
    46	  const step = (halfSize * 2) / PLANE_RENDERING_CONFIG.FALLBACK_GRID.DIVISIONS;
    47	  for (let i = 0; i <= PLANE_RENDERING_CONFIG.FALLBACK_GRID.DIVISIONS; i++) {
    48	    const t = -halfSize + i * step;
    49	    segments.push({ start: { u: -halfSize, v: t }, end: { u: halfSize, v: t } }, { start: { u: t, v: -halfSize }, end: { u: t, v: halfSize } });
    50	  }
