    18	function getCanvasPlaneTransforms(world: World): CanvasPlaneTransformModel[] {
    19	  return world.getAllProjectionPlanes().flatMap((plane) => {
    20	    const realization = world.getProjectionPlane2DRealization(plane.id);
    21	    if (!realization) return [];

    26	function getPlaneCanvasItems(world: World, context: ProjectionElementRealizationContext): CanvasPlaneCanvasItems[] {
    27	  return world.getAllProjectionPlanes().flatMap((plane) => {
    28	    const transform = getCanvasPlaneTransforms(world).find(item => item.planeId === plane.id);
    29	    if (!transform) return [];
    30	    const items: CanvasPlaneItem[] = [];
    31	
    32	    for (const point of plane.getAllPoints()) {
    33	      if (!resolveVisibility(point, world)) continue;
    34	      const realization = plane.realizePoint2D(point, context);
    35	      if (!realization) continue;
    36	      items.push({
    37	        kind: 'point',
    38	        elementId: point.id,
    39	        planeId: point.planeId,
    40	        position: realization.position.clone(),
    41	        pointRole: point.constructor.name === 'FreePoint2D' ? 'free' : 'dependent',
    42	      });
    43	    }
    44	
    45	    for (const line of plane.getAllLines()) {
    46	      if (!resolveVisibility(line, world)) continue;
    47	      const realization = plane.realizeLine2D(line, context);
    48	      if (!realization) continue;
    49	      const kind: CanvasPlaneLineItem['kind'] = line instanceof HalfHingeLine2D
    50	        ? 'half-hinge-line'
    51	        : line instanceof Segment2DByPoints
    52	          ? 'finite-segment'
    53	          : line instanceof SiblingGuide2D
    54	            ? 'sibling-guide'
    55	            : line instanceof Guide2D
    56	              ? 'guide'
    57	              : 'infinite-line';
    58	      items.push({
    59	        kind,
    60	        elementId: line.id,
    61	        planeId: line.planeId,
    62	        point: realization.point1.clone(),
    63	        direction: realization.getDirection(),
    64	        point1: kind === 'finite-segment' ? realization.point1.clone() : undefined,
    65	        point2: kind === 'finite-segment' ? realization.point2.clone() : undefined,
    66	        arcGeometry: optionalArcGeometry(line),
    67	      });
    68	    }
    69	
    70	    for (const circle of plane.getAllCircles()) {
    71	      if (!resolveVisibility(circle, world)) continue;
    72	      const realization = plane.realizeCircle2D(circle, context);
    73	      if (!realization) continue;
    74	      items.push({
    75	        kind: 'circle',
    76	        elementId: circle.id,
    77	        planeId: circle.planeId,
    78	        center: realization.center.clone(),
    79	        radius: realization.getRadius(),
    80	        radiusVector: realization.radiusVector.clone(),
    81	      });
    82	    }
    83	
    84	    return [{ planeId: plane.id, transform, items }];
    85	  });
    86	}

    84	function pointDtoStyle(pointRole: 'free' | 'dependent', planeId: string, ctx: ReturnType<typeof stateFor>): { style: ResolvedStyle; layer: RenderLayer; radius: number } {
    85	  let fill: string; let stroke: string; let strokeWidth: number;
    86	  if (ctx.cascade) { fill = '#FF9800'; stroke = '#F57C00'; strokeWidth = 2.5; }
    87	  else if (ctx.snapTarget) { fill = '#FFA500'; stroke = '#FF8C00'; strokeWidth = 2; }
    88	  else if (ctx.selected) { fill = pointRole === 'free' ? '#4A90E2' : '#E94E77'; stroke = '#0066CC'; strokeWidth = 2; }
    89	  else if (ctx.hovered) { fill = pointRole === 'free' ? '#6BB6FF' : '#FF6B9D'; stroke = '#4A90E2'; strokeWidth = 1.5; }
    90	  else { fill = pointRole === 'free' ? '#4A90E2' : '#E94E77'; stroke = '#333333'; strokeWidth = 1.5; }
    91	  const zBoost = ctx.activePlaneId && planeId === ctx.activePlaneId ? 100 : 0;
    92	  const opacity = ctx.activePlaneId && planeId !== ctx.activePlaneId ? 0.5 : 1;
    93	  return { layer: ctx.snapTarget ? RenderLayer.Highlights : RenderLayer.Elements, radius: pointRole === 'free' ? 5 : 4, style: { stroke, strokeWidth, strokeStyle: 'solid', fill, opacity, zIndex: 10 + zBoost } };
