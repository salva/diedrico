     7	 * Composable for point snapping functionality
     8	 * Finds nearby points and elements within a snap radius
     9	 */
    10	export function useSnapping(snapRadius: Ref<number> = { value: 10 } as Ref<number>) {
    11	  const worldStore = useWorldStore();
    12	  const viewStore = useViewStore();
    13	
    14	  /**
    15	   * Find nearest point to given coordinates within snap radius
    16	   */
    17	  const findNearestPoint = (x: number, y: number): Point2D | null => {
    18	    const activePlaneId = viewStore.activePlaneId;
    19	    if (!activePlaneId) return null;
    20	
    21	    const points = worldStore.allElements2D.filter(
    22	      (el): el is Point2D => el.kind.startsWith('Point2D') && el.planeId === activePlaneId
    23	    );
    24	
    25	    let nearest: Point2D | null = null;
    26	    let minDistance = snapRadius.value;
    27	
    28	    for (const point of points) {
    29	      const px = getPointX(point);
    30	      const py = getPointY(point);
    31	      const distance = Math.sqrt((x - px) ** 2 + (y - py) ** 2);
    32	
    33	      if (distance < minDistance) {
    34	        minDistance = distance;
    35	        nearest = point;
    36	      }
    37	    }
    38	
    39	    return nearest;
    40	  };
    41	
    42	  /**
    43	   * Find all points within snap radius
    44	   */
    45	  const findPointsNear = (x: number, y: number): Point2D[] => {
    46	    const activePlaneId = viewStore.activePlaneId;
    47	    if (!activePlaneId) return [];
    48	
    49	    const points = worldStore.allElements2D.filter(
    50	      (el): el is Point2D => el.kind.startsWith('Point2D') && el.planeId === activePlaneId
    51	    );
    52	
    53	    return points.filter(point => {
    54	      const px = getPointX(point);
    55	      const py = getPointY(point);
    56	      const distance = Math.sqrt((x - px) ** 2 + (y - py) ** 2);
    57	      return distance <= snapRadius.value;
    58	    });
    59	  };

    69	   * Snap coordinates to nearest point if within radius

    76	        y: getPointY(nearest),
    77	        snapped: true
    78	      };
    79	    }
    80	    return { x, y, snapped: false };
    81	  };
    82	
    83	  return {
    84	    findNearestPoint,
    85	    findPointsNear,
    86	    isNearPoint,
    87	    snapToPoint
    88	  };
    89	}

    27	/**
    28	 * Calculate distance from a point to a segment
    29	 */
    30	function distanceToSegment(x: number, y: number, segment: Segment2D, worldStore: any): number {
    31	  if (segment.kind !== 'Segment2DByPoints') return Infinity;
    32	  
    33	  const point1 = worldStore.elements2D.get(segment.point1Id) as Point2D;
    34	  const point2 = worldStore.elements2D.get(segment.point2Id) as Point2D;
    35	  
    36	  if (!point1 || !point2) return Infinity;
    37	  
    38	  const p1 = getPointCoordinates(point1);
    39	  const p2 = getPointCoordinates(point2);
    40	  
    41	  if (!p1 || !p2) return Infinity;
    42	  
    43	  // Vector from p1 to p2
    44	  const dx = p2.x - p1.x;
    45	  const dy = p2.y - p1.y;
    46	  const lengthSquared = dx * dx + dy * dy;
    47	  
    48	  if (lengthSquared === 0) {
    49	    // Degenerate segment (both points are the same)
    50	    return Math.sqrt((x - p1.x) ** 2 + (y - p1.y) ** 2);
    51	  }
    52	  
    53	  // Calculate parameter t (projection onto line)
    54	  let t = ((x - p1.x) * dx + (y - p1.y) * dy) / lengthSquared;
    55	  
    56	  // Clamp t to [0, 1] for segment
    57	  t = Math.max(0, Math.min(1, t));
    58	  
    59	  // Calculate closest point on segment
    60	  const closestX = p1.x + t * dx;
    61	  const closestY = p1.y + t * dy;
    62	  
    63	  // Return distance
    64	  return Math.sqrt((x - closestX) ** 2 + (y - closestY) ** 2);
    65	}

    98	  /**
    99	   * Find the nearest element to a point within a given radius
   100	   */
   101	  const findNearestElement = (x: number, y: number, maxDistance: number = 15): Element2D | null => {
   102	    let nearestElement: Element2D | null = null;
   103	    let minDistance = maxDistance;
   104	    
   105	    for (const element of elementsOnActivePlane.value) {
   106	      let distance = Infinity;
   107	      
   108	      // Skip points (we're looking for lines, segments, circles)
   109	      if (element.kind.startsWith('Point2D')) continue;
   110	      
   111	      if (element.kind === 'Segment2DByPoints') {
   112	        distance = distanceToSegment(x, y, element as Segment2D, worldStore);
   113	      } else if (element.kind.startsWith('Line2D')) {
   114	        distance = distanceToLine(x, y, element as Line2D, worldStore);
   115	      } else if (element.kind.startsWith('Circle2D')) {
   116	        distance = distanceToCircle(x, y, element as Circle2D, worldStore);
   117	      }
   118	      
   119	      if (distance < minDistance) {
   120	        minDistance = distance;
   121	        nearestElement = element;
   122	      }
   123	    }
   124	    
   125	    return nearestElement;
   126	  };

   128	  /**
   129	   * Calculate closest point on an element
   130	   * Returns { x, y, parameter } where parameter is the position along the element
   131	   */
   132	  const closestPointOnElement = (
   133	    x: number, 
   134	    y: number, 
   135	    element: Element2D
   136	  ): { x: number; y: number; parameter: number } | null => {
   137	    if (element.kind === 'Segment2DByPoints') {
   138	      const segment = element as Segment2D;
   139	      const point1 = worldStore.elements2D.get(segment.point1Id) as Point2D;
   140	      const point2 = worldStore.elements2D.get(segment.point2Id) as Point2D;
   141	      
   142	      if (!point1 || !point2) return null;
   143	      
   144	      const p1 = getPointCoordinates(point1);
   145	      const p2 = getPointCoordinates(point2);
   146	      
   147	      if (!p1 || !p2) return null;
   148	      
   149	      // Vector from p1 to p2
   150	      const dx = p2.x - p1.x;
   151	      const dy = p2.y - p1.y;
   152	      const lengthSquared = dx * dx + dy * dy;
   153	      
   154	      if (lengthSquared === 0) return { x: p1.x, y: p1.y, parameter: 0 };
   155	      
   156	      // Calculate parameter t (projection onto line)
   157	      let t = ((x - p1.x) * dx + (y - p1.y) * dy) / lengthSquared;
   158	      
   159	      // Clamp t to [0, 1] for segment
   160	      t = Math.max(0, Math.min(1, t));
   161	      
   162	      // Calculate closest point
   163	      const closestX = p1.x + t * dx;
   164	      const closestY = p1.y + t * dy;
   165	      
   166	      return { x: closestX, y: closestY, parameter: t };
   167	    }
   168	    
   169	    // TODO: Implement for lines and circles
   170	    return null;
   171	  };

   173	  /**
   174	   * Find elements that intersect at or near a point
   175	   */
   176	  const findIntersectingElements = (
   177	    x: number, 
   178	    y: number, 
   179	    maxDistance: number = 15
   180	  ): Array<{ element1: Element2D; element2: Element2D }> => {
   181	    const intersections: Array<{ element1: Element2D; element2: Element2D }> = [];
   182	    
   183	    // Find all elements near the point
   184	    const nearbyElements = elementsOnActivePlane.value.filter((el: Element2D) => {
   185	      if (el.kind.startsWith('Point2D')) return false;
   186	      
   187	      let distance = Infinity;
   188	      if (el.kind === 'Segment2DByPoints') {
   189	        distance = distanceToSegment(x, y, el as Segment2D, worldStore);
   190	      } else if (el.kind.startsWith('Line2D')) {
   191	        distance = distanceToLine(x, y, el as Line2D, worldStore);
   192	      } else if (el.kind.startsWith('Circle2D')) {
   193	        distance = distanceToCircle(x, y, el as Circle2D, worldStore);
   194	      }
   195	      
   196	      return distance < maxDistance;
   197	    });
   198	    
   199	    // Find pairs of elements that intersect
   200	    for (let i = 0; i < nearbyElements.length; i++) {
   201	      for (let j = i + 1; j < nearbyElements.length; j++) {
   202	        // TODO: Calculate actual intersection and check if it's near (x, y)
   203	        // For now, just return all pairs
   204	        intersections.push({
   205	          element1: nearbyElements[i],
   206	          element2: nearbyElements[j]
   207	        });
   208	      }
   209	    }
   210	    
   211	    return intersections;
   212	  };
   213	  
   214	  return {
   215	    findNearestElement,
   216	    closestPointOnElement,
   217	    findIntersectingElements,
   218	    elementsOnActivePlane
   219	  };

### World ownership
     1	/**
     2	 * World - Root container for all geometric elements
     3	 * 
     4	 * The World class stores all geometric entities (points, lines, segments, circles, planes, etc.)
     5	 * in the dihedral drawing system. It serves as the central data structure that holds the 
     6	 * complete state of a construction.
     7	 * 
     8	 * Design principles (from Constitution):
     9	 * - Principle I: Domain layer - pure business logic, no UI dependencies
    10	 * - Principle V: All points stored in plane intrinsic coordinates (homogeneous)
    11	 * 
    12	 * ## Architecture: Element Access Patterns (Phase 48)
    13	 * 
    14	 * **Ownership Hierarchy**: World → ProjectionPlane → Element2D
    15	 * - 2D elements (points, lines, circles, etc.) are stored in and owned by ProjectionPlane
    16	 * - World provides high-level access methods but does NOT store 2D elements directly
    17	 * - This ensures proper encapsulation and clear ownership boundaries
    18	 * 
    19	 * **How to Access Elements**:
    20	 * 
    21	 * 1. **Via Hierarchical ID** (Phase 45 - preferred for known IDs):
    22	 *    ```typescript
    23	 *    const point = world.getPoint2D(pointId);        // Returns Point2D | undefined
    24	 *    const line = world.getLine2D(lineId);           // Returns Line2D | undefined
    25	 *    const circle = world.getCircle2D(circleId);     // Returns Circle2D | undefined
    26	 *    const element = world.getElement2D(elementId);  // Returns Element2D | undefined
    27	 *    ```
    28	 *    These methods use hierarchical IDs (e.g., "proj_plane_1:point_5") to find the 
    29	 *    correct plane and retrieve the element. Type-safe and fast.
    30	 * 
    31	 * 2. **Via ProjectionPlane** (when you already have the plane):
    32	 *    ```typescript
    33	 *    const plane = world.getProjectionPlaneById(planeId);
    34	 *    const point = plane.getPoint(pointId);
    35	 *    const line = plane.getLine(lineId);
    36	 *    ```
    37	 * 
    38	 * 3. **Getting Realizations** (computed geometric data):
    39	 *    ```typescript
    40	 *    const plane = world.getProjectionPlaneById(planeId);
    41	 *    const point = plane.getPoint(pointId);
    42	 *    const realization = point?.realize2D(world);  // Get computed coordinates
    43	 *    ```
    44	 *    Realizations contain the actual computed positions/geometry. The domain object
    45	 *    stores the construction definition (e.g., "intersection of line A and circle B"),
    46	 *    while the realization stores the computed result (e.g., actual [x, y] coordinates).
    47	 * 
    48	 * 4. **Iterating All Elements** (for operations like select-all, export):
    49	 *    ```typescript
    50	 *    for (const plane of world.getAllProjectionPlanes()) {
    51	 *      for (const point of plane.getAllPoints()) {
    52	 *        // Process each point
    53	 *      }
    54	 *    }
    55	 *    ```
    56	 * 
    57	 * 5. **Semantic Access** (Phase 46 - for fixed initial planes):
    58	 *    ```typescript
    59	 *    const horizontalPlane = world.getProjectionPlaneByLabel('P_H');
    60	 *    const verticalPlane = world.getProjectionPlaneByLabel('P_V');
    61	 *    const originH = horizontalPlane.getPoint2DByLabel('O_H');  // Phase 53
    62	 *    const hingeLine = world.getMainHingeLine();
    63	 *    ```
    64	 * 
    65	 * **Why Facade Methods Were Removed** (Phase 48):
    66	 * 
    67	 * Previous versions had convenience methods like `world.getPoint2DRealization(id)` that
    68	 * bypassed ProjectionPlane ownership. These were removed because:
    69	 * - **Violated encapsulation**: World shouldn't know about ProjectionPlane's internal storage
    70	 * - **Tight coupling**: Client code became dependent on World instead of ProjectionPlane
    71	 * - **Inconsistent patterns**: Some code used facade, some used proper ownership
    72	 * - **Maintenance burden**: World had to mirror ProjectionPlane's interface
    73	 * 
    74	 * The current architecture enforces proper ownership: ProjectionPlane owns its elements,
    75	 * World provides semantic access and ID-based lookup.
    76	 * 
    77	 * @see specs/001-dihedral-learning-mvp/data-model.md#world-entity
    78	 * @see Phase 45: Hierarchical Element IDs
    79	 * @see Phase 46: Semantic Accessors
    80	 * @see Phase 48: Remove World Facade Methods
    81	 */
