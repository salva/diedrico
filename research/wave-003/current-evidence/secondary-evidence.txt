     7	 * Service for validation and infeasibility checks

    11	  private static readonly COLLINEARITY_THRESHOLD = 1e-10;
    12	
    13	  /**
    14	   * Validate an element for infeasibility

    21	      return Err(InfeasibilityReason.MissingDependency);

    26	
    27	  /**
    28	   * Check if two points are coincident
    29	   */
    30	  static checkCoincidentPoints(
    31	    point1: { x: number; y: number },
    32	    point2: { x: number; y: number }
    33	  ): Result<void, InfeasibilityReason> {
    34	    if (isWithinDistance(point1, point2, this.COINCIDENCE_THRESHOLD)) {
    35	      return Err(InfeasibilityReason.CoincidentPoints);
    36	    }
    37	    return Ok(undefined);
    38	  }
    39	
    40	  /**
    41	   * Check if three points are collinear
    42	   */
    43	  static checkCollinearity(
    44	    p1: { x: number; y: number },
    45	    p2: { x: number; y: number },
    46	    p3: { x: number; y: number }
    47	  ): Result<void, InfeasibilityReason> {
    48	    // Calculate cross product to check collinearity
    49	    // (p2 - p1) × (p3 - p1) = 0 for collinear points
    50	    const crossProduct = 
    51	      (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
    52	    
    53	    if (Math.abs(crossProduct) < this.COLLINEARITY_THRESHOLD) {
    54	      return Err(InfeasibilityReason.CollinearPoints);
    55	    }
    56	    
    57	    return Ok(undefined);
    58	  }
    59	
    60	  /**
    61	   * Check if two lines are parallel
    62	   */
    63	  static checkParallelLines(
    64	    line1: { x1: number; y1: number; x2: number; y2: number },
    65	    line2: { x1: number; y1: number; x2: number; y2: number }
    66	  ): Result<void, InfeasibilityReason> {
    67	    const dx1 = line1.x2 - line1.x1;
    68	    const dy1 = line1.y2 - line1.y1;
    69	    const dx2 = line2.x2 - line2.x1;
    70	    const dy2 = line2.y2 - line2.y1;
    71	    
    72	    // Check if cross product is near zero (parallel lines)
    73	    const crossProduct = dx1 * dy2 - dy1 * dx2;
    74	    
    75	    if (Math.abs(crossProduct) < this.COLLINEARITY_THRESHOLD) {
    76	      return Err(InfeasibilityReason.ParallelLines);
    77	    }
    78	    
    79	    return Ok(undefined);
    80	  }
